Шифр Юлия

Юлий Цезарь использовал свой способ шифрования текста. Каждая буква заменялась на следующую по алфавиту через K позиций по кругу. Необходимо по заданной шифровке определить исходный текст.

Входные данные
В первой строке дана шифровка, состоящая из заглавных латинских букв. Во второй строке число K (1 ≤ K ≤ 10).

Выходные данные
Требуется вывести результат расшифровки.

Sample Input 1:
XPSE
1

Sample Output 1:
WORD

Sample Input 2:
ZABC
3

Sample Output 2:
WXYZ

# python
def f(text: str, n: int) -> str:
    res = ''
    for letter in text:
        res += chr((ord(letter) - 97 - n % 26) % 26 + 97)
    return res

print(f(input().lower(), int(input())).upper())

use std::io::{self, Read};

/// Shifts each lowercase ASCII letter in `text` backwards by `shift` positions
/// in the alphabet (wrapping around). Non‑alphabetic characters are left unchanged.
fn shift_backwards(text: &str, shift: usize) -> String {
    let mut result = String::with_capacity(text.len());
    let effective_shift = shift % 26; // rotation within the alphabet

    for ch in text.chars() {
        if ('a'..='z').contains(&ch) {
            // Convert to 0‑based index, apply shift, wrap with Euclidean remainder.
            let shifted_index = ((ch as u8 - b'a') as i32 - effective_shift as i32)
                .rem_euclid(26) as u8 + b'a';
            result.push(shifted_index as char);
        } else {
            // Preserve any character that is not a lowercase letter.
            result.push(ch);
        }
    }

    result
}

fn main() {
    // Read the entire stdin (two lines: the text and the integer).
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut lines = input.lines();

    // First line: the text to process, converted to lowercase.
    let raw_text = lines.next().unwrap_or("").to_lowercase();

    // Second line: the shift amount, parsed as usize.
    let shift_amount: usize = lines
        .next()
        .unwrap_or("0")
        .trim()
        .parse()
        .unwrap_or(0);

    // Apply the cipher and output the result in uppercase.
    let shifted = shift_backwards(&raw_text, shift_amount);
    println!("{}", shifted.to_uppercase());
}
