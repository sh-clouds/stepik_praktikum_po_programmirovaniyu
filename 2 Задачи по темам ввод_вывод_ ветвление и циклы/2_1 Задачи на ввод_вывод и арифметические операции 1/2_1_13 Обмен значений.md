Обмен значений

Напишите программу, которая считывает значения двух целочисленных переменных a и b, затем меняет их значения местами (то есть в переменной a должно быть записано то, что раньше хранилось в b, а в переменной b записано то, что раньше хранилось в a). Затем выведите значения переменных.

Входные данные
Вводятся целые числа a и b.

Выходные данные
Выведите ответ на задачу.

Sample Input:
8
5

Sample Output:
5 8

# C
#include <stdio.h>

int main() {
    int a, b; 
    scanf("%d %d", &a, &b);
    
    swap_int(&a, &b);
    
    printf("%d %d\n", a, b);
    return 0;
}

void swap_int(int *a, int *b) {
    int t = *a;
    *a = *b;
    *b = t;
}

# rust
use std::io::{self, Read};

fn main() {
    // Read the entire stdin into a string and split by lines
    let mut input = String::new();
    io::stdin()
        .read_to_string(&mut input)
        .expect("Failed to read from stdin");

    // Collect non‑empty lines (mirrors Python's input() behavior)
    let mut lines = input.lines();

    let first_input = lines.next().unwrap_or("").trim_end();
    let second_input = lines.next().unwrap_or("").trim_end();

    // Print in swapped order, matching Python's `print(b, a)`
    println!("{} {}", second_input, first_input);
}
