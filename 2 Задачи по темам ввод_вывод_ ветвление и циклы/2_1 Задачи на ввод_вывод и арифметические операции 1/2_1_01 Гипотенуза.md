Гипотенуза

Дано два числа a и b. Найдите гипотенузу треугольника с заданными катетами.

Входные данные
В двух строках вводятся два числа (числа целые, положительные, не превышают 1000).

Выходные данные
Выведите ответ на задачу в виде целого числа

Sample Input:
6
8

Sample Output:
10

# rust
use std::io::{self, BufRead};

/// Computes the length of the hypotenuse of a right‑angled triangle
/// given the lengths of the other two sides `a` and `b`.
/// The result is rounded to the nearest integer.
fn triangle(a: i32, b: i32) -> i32 {
    // Convert to floating point for the square‑root calculation,
    // then round and cast back to `i32`.
    ((a as i64 * a as i64 + b as i64 * b as i64) as f64)
        .sqrt()
        .round() as i32
}

fn main() {
    // Read two lines from standard input.
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let a: i32 = lines
        .next()
        .expect("expected first input line")
        .unwrap()
        .trim()
        .parse()
        .expect("first input must be an integer");

    let b: i32 = lines
        .next()
        .expect("expected second input line")
        .unwrap()
        .trim()
        .parse()
        .expect("second input must be an integer");

    println!("{}", triangle(a, b));
}
