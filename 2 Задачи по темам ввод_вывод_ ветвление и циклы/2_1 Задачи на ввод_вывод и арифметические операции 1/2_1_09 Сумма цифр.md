# rust
use std::io::{self, Read};

fn main() {
    // Read the whole input from stdin
    let mut raw_input = String::new();
    io::stdin().read_to_string(&mut raw_input).unwrap();

    // Remove any trailing newline characters
    let trimmed_input = raw_input.trim();

    // Remove all hyphens from the input
    let digits_only: String = trimmed_input.chars().filter(|c| *c != '-').collect();

    // Convert each character to a digit and sum them
    let total_sum: i32 = digits_only
        .chars()
        .filter_map(|c| c.to_digit(10))   // Keep only numeric characters
        .map(|d| d as i32)                // Convert u32 to i32 for summing
        .sum();

    // Output the result
    println!("{}", total_sum);
}
