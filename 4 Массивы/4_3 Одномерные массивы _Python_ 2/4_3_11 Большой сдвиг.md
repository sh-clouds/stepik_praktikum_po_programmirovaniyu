Большой сдвиг

Дан список из N (1≤N≤100000) целых чисел и число K (|K|<100000). Циклически сдвиньте список на |K| элементов вправо, если K – положительное и влево, если отрицательное число.

Входные данные
Программа получает на вход список целых чисел, затем число K.

Выходные данные
Выведите ответ на задачу.

Примечание
Решение должно иметь сложность O(N), то есть не должно зависеть от K. Дополнительным списком пользоваться нельзя.

Sample Input:
5 3 7 4 6
3

Sample Output:
7 4 6 5 3

#include <inttypes.h>

#include <stdlib.h>

void input(int64_t** arr, size_t* N, int64_t* K);
void output(int64_t** arr, size_t N);

int8_t puton_check(size_t start, size_t checkpoint, int64_t K, size_t N);

int main() {
    size_t N = 0;
    int64_t K = 0;
    int64_t* array;
    input(&array, &N, &K);

    while(K < 0) K += N;
    puton(0, 0, K, N, &array);
    for (size_t i = 1; i < N; i++) {
        int8_t visited = 0;
        for (size_t j = 0; j < i; j++) {
            visited = puton_check(j, i, K, N);
            if (visited) break;
        }
        if (!visited) puton(i, i, K, N, &array);
    }

    output(&array, N);
    if (N) free(array);
    return 0;
}

int8_t puton_check(size_t start, size_t checkpoint, int64_t K, size_t N) {
    int8_t visited = 0;
    size_t current = start, next;
    do {
        next = (current + K) % N;
        if (checkpoint == next) {
            visited = 1;
            break;
        }
        current = next;
    } while (next != start);
    return visited;
}

void puton(size_t start, size_t current, int64_t K, size_t N, int64_t** arr) {
    int64_t item_current = (*arr)[current];
    size_t next = (current + K) % N;
    if (next != start) puton(start, next, K, N, &(*arr));
    (*arr)[next] = item_current;
}

void output(int64_t** arr, size_t N) {
    for (size_t i = 0; i < N; i++) {
        printf("%"PRId64, (*arr)[i]);
        if (i != N - 1) printf(" ");
    }
}

void input(int64_t** arr, size_t* N, int64_t* K) {
    int64_t num;
    (*N) = 0;
    while (scanf("%"SCNd64, &num) == 1) {
        (*N)++;

        else (*arr) = realloc((*arr), (*N) * sizeof(int64_t));

    }
    (*K) = (*arr)[(*N) - 1];
    (*N)--;
    (*arr) = realloc((*arr), (*N) * sizeof(int64_t));
}

# rust
use std::collections::VecDeque;
use std::io::{self, Read};

fn main() {
    // Read the whole input
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    // Split into lines: first line = list of integers, second line = delta
    let mut lines = input.lines();

    // Parse the list of integers
    let first_line = lines.next().unwrap_or("");
    let numbers: Vec<i32> = first_line
        .split_whitespace()
        .map(|s| s.parse::<i32>().expect("invalid integer"))
        .collect();

    // Parse the rotation delta
    let second_line = lines.next().unwrap_or("0");
    let delta: isize = second_line
        .trim()
        .parse::<isize>()
        .expect("invalid delta");

    // Build a VecDeque from the numbers
    let mut deque: VecDeque<i32> = VecDeque::from(numbers);
    let len = deque.len();

    // Perform rotation if the deque is not empty
    if len > 0 {
        // Normalize delta to a value in [0, len)
        let rot = ((delta % len as isize) + len as isize) % len as isize;
        if rot > 0 {
            deque.rotate_right(rot as usize);
        }
    }

    // Print the resulting deque, space‑separated
    let output = deque.iter()
        .map(|v| v.to_string())
        .collect::<Vec<_>>()
        .join(" ");
    println!("{}", output);
}
