Послезавтра

По заданной дате требуется определить, какое число будет послезавтра.
Напомним, что год является високосным, если его номер кратен 4, но не кратен 100, а также если он кратен 400.

Входные данные
Дано число, месяц и год (год  – число в промежутке от 1 до 10000).

Выходные данные
Требуется вывести, какое число будет послезавтра, в формате входных данных.

Sample Input:
1 8 2009

Sample Output:
3 8 2009

Напишите программу. Тестируется через stdin → stdout

use std::io::{self, Read};

/// Determine whether a given year is a leap year.
fn is_leap_year(year: i32) -> bool {
    (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)
}

/// Return the number of days in a month for a specific year (1‑based month index).
fn days_in_month(month: usize, year: i32) -> i32 {
    // Days per month for a common year.
    const COMMON: [i32; 12] = [31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];
    // Days per month for a leap year.
    const LEAP:   [i32; 12] = [31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31];

    if is_leap_year(year) {
        LEAP[month - 1]
    } else {
        COMMON[month - 1]
    }
}

fn main() {
    // ----- read input ---------------------------------------------------------
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).expect("failed to read stdin");
    // Keep only the first non‑empty line
    let first_line = input
        .lines()
        .map(str::trim)
        .find(|line| !line.is_empty())
        .expect("no input data");

    // Parse day, month and year (the Python code expects them space‑separated)
    let mut parts = first_line.split_whitespace();
    let day: i32 = parts
        .next()
        .expect("missing day")
        .parse()
        .expect("day is not an integer");
    let month: usize = parts
        .next()
        .expect("missing month")
        .parse()
        .expect("month is not an integer");
    let year: i32 = parts
        .next()
        .expect("missing year")
        .parse()
        .expect("year is not an integer");

    // ----- add two days -------------------------------------------------------
    let mut new_day = day + 2;
    let mut new_month = month;
    let mut new_year = year;

    // Adjust day/month/year if the addition overflows the current month/year
    loop {
        let dim = days_in_month(new_month, new_year);
        if new_day <= dim {
            break;
        }
        new_day -= dim;
        new_month += 1;
        if new_month > 12 {
            new_month = 1;
            new_year += 1;
        }
    }

    // ----- output -------------------------------------------------------------
    println!("{} {} {}", new_day, new_month, new_year);
}
