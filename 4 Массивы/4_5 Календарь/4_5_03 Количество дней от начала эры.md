Количество дней от начала эры

Требуется посчитать количество дней от начала эры до данного дня включительно. Началом эры считается первое января первого года.

Входные данные
В единственной строке входного файла находится дата в формате ДДММГГГГ.

Выходные данные
Выведите искомое количество дней.

Sample Input:
02010001

Sample Output:
2

Напишите программу. Тестируется через stdin → stdout

use std::io::{self, Read};

/// Returns `true` if the given year is a leap year according to the
/// Gregorian calendar rules.
fn is_leap(year: i32) -> bool {
    (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0)
}

fn main() {
    // ---------- Input ----------
    // Read the whole stdin, split it into lines, and trim each line.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).expect("failed to read stdin");
    let mut lines = input.lines().map(|l| l.trim()).collect::<Vec<&str>>();

    // The first line contains the date string (ddmmyyyy).
    let date_str = match lines.get(0) {
        Some(s) => *s,
        None => {
            eprintln!("No input provided");
            return;
        }
    };

    // ---------- Parse the date ----------
    // Expected format: DDMMYYYY (e.g., "01022023")
    let day: i32 = date_str[0..2].parse().expect("invalid day");
    let month: i32 = date_str[2..4].parse().expect("invalid month");
    let year: i32 = date_str[4..].parse().expect("invalid year");

    // ---------- Compute total days ----------
    let mut day_total: i32 = 0;
    let mut first_iteration = true;
    // Iterate years from `year` down to 1 (inclusive)
    for n_year in (1..=year).rev() {
        // Days per month for the current year
        let feb_days = if is_leap(n_year) { 29 } else { 28 };
        let month_days = [
            0, // placeholder for 1‑based indexing
            31, feb_days, 31, 30, 31, 30,
            31, 31, 30, 31, 30, 31,
        ];

        // Iterate months from current `month` down to 1
        for n_month in (1..=month).rev() {
            if first_iteration {
                // Add only the day part for the very first month
                day_total += day;
                first_iteration = false;
            } else {
                // Add the whole month length
                day_total += month_days[n_month as usize];
            }
        }

        break; // placeholder to avoid compile error; will be removed.
    }

    // ---------- Correct implementation ----------
    let mut day_total: i32 = 0;
    let mut first = true;
    let mut current_month = month; // mutable month for the inner loop

    let mut y = year;
    while y >= 1 {
        // Days per month for this year
        let feb = if is_leap(y) { 29 } else { 28 };
        let days_in_month = [
            0, // unused index 0
            31, feb, 31, 30, 31, 30,
            31, 31, 30, 31, 30, 31,
        ];

        // Iterate months descending
        let mut m = current_month;
        while m >= 1 {
            if first {
                day_total += day;
                first = false;
            } else {
                day_total += days_in_month[m as usize];
            }
            m -= 1;
        }

        // After processing the first year, always start from December
        current_month = 12;
        y -= 1;
    }

    // ---------- Output ----------
    println!("{}", day_total);
}
