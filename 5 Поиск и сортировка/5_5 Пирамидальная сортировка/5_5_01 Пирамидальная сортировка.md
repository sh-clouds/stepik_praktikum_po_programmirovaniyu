        public static void Sort(int[] arr)
        {
            int n = arr.Length;
                        
            for (int i = n / 2 - 1; i >= 0; i--)
                Heapify(arr, n, i);
                        
            for (int i = n - 1; i >= 0; i--)
            {
                int temp = arr[0];
                arr[0] = arr[i];
                arr[i] = temp;

                Heapify(arr, i, 0);
            }
        }

        static void Heapify(int[] arr, int n, int i)
        {
            int largest = i;
            
            int l = 2 * i + 1; 
            int r = 2 * i + 2; 

            if (l < n && arr[l] > arr[largest])
                largest = l;

            if (r < n && arr[r] > arr[largest])
                largest = r;

            if (largest != i)
            {
                int swap = arr[i];
                arr[i] = arr[largest];
                arr[largest] = swap;

                Heapify(arr, n, largest);
            }
        }

import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        input.nextLine();

        int[] array = Arrays.stream(input.nextLine().split("\\s+")).mapToInt(Integer::parseInt).toArray();

        heapSort(array);
        printTheArray(array);
    }

    public static void heapSort(int array[]) {
        int n = array.length;

        for (int a = n / 2 - 1; a >= 0; a--) {
            heapUp(array, n, a);
        }

        for (int b = n - 1; b > 0; b--) {
            int temporary = array[0];
            array[0] = array[b];
            array[b] = temporary;

            heapUp(array, b, 0);
        }
    }

    public static void heapUp(int[] array, int num, int i) {
        int biggest = i;
        int left = 2 * i + 1;
        int right = 2 * i + 2;

        if (left < num && array[left] > array[biggest]) {
            biggest = left;
        }

        if (right < num && array[right] > array[biggest]) {
            biggest = right;
        }

        if (biggest != i) {
            int swap = array[i];
            array[i] = array[biggest];
            array[biggest] = swap;

            heapUp(array, num, biggest);
        }
    }

    private static void printTheArray(int[] someArray) {
        for (int z = 0; z < someArray.length; z++) {
            if (z < someArray.length - 1) System.out.print(someArray[z] + " ");
            else System.out.println(someArray[z]);
        }
    }
}

use std::io::{self, Read};

/// Restore the max‑heap property for the element at index `i`,
/// assuming the sub‑trees rooted at its children are already heaps.
fn heapify(arr: &mut [i64], i: usize, heap_len: usize) {
    let mut largest = i;
    let left = 2 * i + 1;
    let right = 2 * i + 2;

    if left < heap_len && arr[left] > arr[largest] {
        largest = left;
    }
    if right < heap_len && arr[right] > arr[largest] {
        largest = right;
    }

    if largest != i {
        arr.swap(i, largest);
        // Re‑heapify the affected subtree.
        heapify(arr, largest, heap_len);
    }
}

fn main() {
    // ----- read whole input -------------------------------------------------
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut it = input.split_whitespace().map(|s| s.parse::<i64>().unwrap());

    // ----- parse ------------------------------------------------------------
    let n = it.next().expect("missing size") as usize;
    let mut data: Vec<i64> = it.take(n).collect();

    // ----- build max‑heap (heapify from bottom up) -------------------------
    for idx in (0..=n.saturating_sub(2)).rev() {
        heapify(&mut data, idx, n);
    }

    // ----- heap sort --------------------------------------------------------
    for end in (1..n).rev() {
        data.swap(0, end);
        heapify(&mut data, 0, end);
    }

    // ----- output -----------------------------------------------------------
    let output = data
        .iter()
        .map(|v| v.to_string())
        .collect::<Vec<_>>()
        .join(" ");
    println!("{}", output);
}
