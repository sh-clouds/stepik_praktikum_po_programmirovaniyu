def sortingByCount(m)
    k = m.max 
    scope = k + 1
    c = [0] * scope
    
    0.upto(m.size-1) do |i|
        c[m[i]] += 1
    end  
  
    m = []    
    0.upto(k) do |j|
        0.upto(c[j]-1) do |i|
            m.push(j)
        end
    end

    return m     
end

def sortingByCount(A):
    scope = max(A) + 1
    C = [0] * scope
    for x in A:
        C[x] += 1
    A = []
    for number in range(scope):
        A += [number] * C[number] 
    return A

import java.util.Arrays;
import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int number = Integer.parseInt(input.nextLine());
        int[] array = IntStream.generate(input::nextInt).limit(number).toArray();

        array = countingSort(array);
        printTheArray(array);
    }

    private static int[] countingSort(int[] array) {
        int max = array[0], min = array[0], counter = 0;

        for (int a = 1; a < array.length; a++) {
            if(array[a] > max) max = array[a];
            if(array[a] < min) min = array[a];
        }

        int range = max - min + 1;
        int[] temporary = new int[range];

        for (int b = 0; b < array.length; b++) {
            if(array[b] == 0) temporary[Math.abs(min)] += 1;
            else temporary[array[b] - min] += 1;
        }

        for (int e = 0; e < temporary.length; e++) {
            if(temporary[e] > 0) {
                for (int f = 0; f < temporary[e]; f++) {
                    array[counter] = e + min;
                    counter++;
                }
            }
        }

        return array;
    }

    private static void printTheArray(int[] array) {
        for (int a = 0; a < array.length; a++) {
            if (a < array.length - 1) System.out.print(array[a] + " ");
            else System.out.println(array[a]);
        }
    }
}

use std::io::{self, Read};

/// Performs counting sort on a vector of non‑negative integers.
/// Returns a new vector containing the sorted elements.
fn counting_sort(non_negative: Vec<i32>) -> Vec<i32> {
    // Find the maximum value to determine the size of the counting array.
    let max_value = *non_negative.iter().max().unwrap();
    let range = (max_value as usize) + 1;

    // Count occurrences of each number.
    let mut frequency = vec![0usize; range];
    for &value in &non_negative {
        frequency[value as usize] += 1;
    }

    // Reconstruct the sorted list based on the frequencies.
    let mut sorted = Vec::with_capacity(non_negative.len());
    for number in 0..range {
        for _ in 0..frequency[number] {
            sorted.push(number as i32);
        }
    }
    sorted
}

/// Prints the contents of an integer vector separated by spaces,
/// followed by a newline.
fn print_vector(values: Vec<i32>) {
    let mut first = true;
    for v in values {
        if !first {
            print!(" ");
        }
        print!("{}", v);
        first = false;
    }
    println!();
}

fn main() {
    // Read the entire stdin input.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut tokens = input.split_whitespace();

    // The original Python code discards the first line (e.g., a count).
    // We simply skip the first token if it exists.
    tokens.next();

    // Parse the remaining tokens as integers.
    let mut numbers: Vec<i32> = tokens.map(|t| t.parse::<i32>().unwrap()).collect();

    if numbers.is_empty() {
        // Nothing to sort; exit early.
        return;
    }

    // Determine the minimal value to decide whether an offset is needed.
    let min_value = *numbers.iter().min().unwrap();

    if min_value < 0 {
        // Offset all numbers to make them non‑negative.
        let offset = -min_value;
        let shifted: Vec<i32> = numbers.iter().map(|&x| x + offset).collect();

        // Sort the shifted values and then remove the offset.
        let sorted_shifted = counting_sort(shifted);
        let restored: Vec<i32> = sorted_shifted.iter().map(|&x| x - offset).collect();

        print_vector(restored);
    } else {
        // All numbers are already non‑negative.
        let sorted = counting_sort(numbers);
        print_vector(sorted);
    }
}
