import java.util.Scanner;
import java.util.stream.IntStream;

public class Main {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        int number = Integer.parseInt(input.nextLine());
        int[] array = IntStream.generate(input::nextInt).limit(number).toArray();
        mergeSort(array, 0, array.length -1);
        printTheArray(array);
    }

    private static void mergeSort(int[] array, int left, int right) {
        if (left < right) {
            int middlePoint = (left + right) / 2;

            mergeSort(array, left, middlePoint);
            mergeSort(array, middlePoint + 1, right);
            merge(array, left, middlePoint, right);
        }
    }

    private static void merge(int[] array, int left, int middlePoint, int right) {
        int size1 = middlePoint - left + 1;
        int size2 = right - middlePoint;

        int[] theFirst = new int[size1];
        int[] theSecond = new int[size2];

        for (int a = 0; a < size1; ++a) {
            theFirst[a] = array[left + a];
        }

        for (int b = 0; b < size2; ++b) {
            theSecond[b] = array[middlePoint + 1 + b];
        }

        int e = 0, f = 0, u = left;

        while (e < size1 && f < size2) {
            if (theFirst[e] <= theSecond[f]) {
                array[u] = theFirst[e];
                e++;
            }
            else {
                array[u] = theSecond[f];
                f++;
            }
            u++;
        }

        while (e < size1) {
            array[u] = theFirst[e];
            e++;
            u++;
        }
        
        while (f < size2) {
            array[u] = theSecond[f];
            f++;
            u++;
        }
    }

    private static void printTheArray(int[] array) {
        for (int p = 0; p < array.length; p++) {
            if (p < array.length - 1) System.out.print(array[p] + " ");
            else System.out.println(array[p]);
        }
    }
}

void MergeSorting(vector<int>& arr, int start, int end) {
    int mid;
    if (start < end) {
        mid = (start + end) / 2;
        MergeSorting(arr, start, mid);
        MergeSorting(arr, mid + 1, end);

        vector<int> sortArr(arr.size());
        int i = start;
        int j = mid + 1;
        int k = start;

        while (i <= mid && j <= end) {
            if (arr[i] < arr[j]) sortArr[k++] = arr[i++];
            else sortArr[k++] = arr[j++];
        }

        while (i <= mid) sortArr[k++] = arr[i++];
        while (j <= end) sortArr[k++] = arr[j++];

        for (int i = start; i < k; i++) arr[i] = sortArr[i];
    }
}

use std::io::{self, Read};

/// Perform the Bose‑Nelson sorting algorithm on the provided vector.
fn bose_nelson(data: &mut Vec<i32>) {
    let mut m = 1usize;
    while m < data.len() {
        let mut j = 0usize;
        while j + m < data.len() {
            bose_nelson_merge(j, m, m, data);
            j += m + m;
        }
        m += m;
    }
}

/// Recursive helper that merges two sub‑sequences according to the
/// Bose‑Nelson method.
///
/// * `j` – start index of the first sub‑sequence  
/// * `r` – distance to the start of the second sub‑sequence  
/// * `m` – current size of the sub‑sequences being merged
fn bose_nelson_merge(j: usize, r: usize, m: usize, data: &mut Vec<i32>) {
    if j + r >= data.len() {
        return;
    }

    if m == 1 {
        if data[j] > data[j + r] {
            data.swap(j, j + r);
        }
    } else {
        let half = m / 2;
        // First half‑size merge
        bose_nelson_merge(j, r, half, data);

        // Conditional second merge if the range fits
        if j + r + half < data.len() {
            bose_nelson_merge(j + half, r, half, data);
        }

        // Final merge with reduced distance
        bose_nelson_merge(j + half, r - half, half, data);
    }
}

fn main() {
    // ----- Read complete stdin -----
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut iter = input.split_whitespace();

    // First value is the length (unused, but consumed)
    let _n: usize = iter
        .next()
        .expect("expected length")
        .parse()
        .expect("length must be an integer");

    // Remaining values constitute the list to sort
    let mut numbers: Vec<i32> = iter
        .map(|s| s.parse::<i32>().expect("all items must be integers"))
        .collect();

    // ----- Sorting -----
    bose_nelson(&mut numbers);

    // ----- Output -----

        .iter()

        .collect::<Vec<_>>()
        .join(" ");
    println!("{}", output);
}
