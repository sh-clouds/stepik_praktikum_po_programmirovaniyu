Входные данные
В первой строке дано число N – количество чисел. (1 <= N <= 100000) Во второй строке даны через пробел N чисел, каждое не превышает 2*109 по модулю.

Выходные данные
Выведите число, равное количеству различных чисел среди данных.

Sample Input:
5
1 0 1 2 0

Sample Output:
3

# python
def quick_sort(lst):
    if len(lst) < 2:
        return lst
    mid = lst[len(lst) // 2]
    center = [n for n in lst if n == mid]
    left = list(filter(lambda n: n < mid, lst))
    right = list(filter(lambda n: n > mid, lst))
    
    return quick_sort(left) + center + quick_sort(right)

# C++
#include <iostream>
#include <vector>
using namespace std;

void QuickSorting(vector<int>& arr, int start, int end) {
    int i = start;
    int pivot = end;
    while (i < pivot) {
        if (arr[i] > arr[pivot] && i == pivot - 1) {
            swap(arr[i], arr[pivot]);
            pivot--;
        }
        else if (arr[i] > arr[pivot]) {
            swap(arr[pivot], arr[pivot - 1]);
            swap(arr[i], arr[pivot]);
            pivot--;
        }
        else i++;
    }
    if (start < end) {
        QuickSorting(arr, start, pivot - 1);
        QuickSorting(arr, pivot + 1, end);
    }
}

int main() {
    int N; cin >> N;
    vector<int> arr(N);
    for (int i = 0; i < N; i++) cin >> arr[i];
    QuickSorting(arr, 0, N - 1);

    int unique = 1;
    for (int i = 1; i < N; i++) {
        if (arr[i] != arr[i - 1]) unique++;
    }
    cout << unique <<  "\n\n";
    return 0;
}


# rust
use std::collections::HashSet;
use std::io::{self, Read};

/// Recursively sorts a vector of integers using the quick‑sort algorithm.
fn quick_sort(mut list: Vec<i32>) -> Vec<i32> {
    if list.len() < 2 {
        return list;
    }

    // Choose the middle element as the pivot.
    let pivot = list[list.len() / 2];

    // Partition the list into three groups.
    let mut left = Vec::new();
    let mut middle = Vec::new();
    let mut right = Vec::new();

    for &value in &list {
        if value < pivot {
            left.push(value);
        } else if value == pivot {
            middle.push(value);
        } else {
            right.push(value);
        }
    }

    // Recursively sort the left and right partitions and concatenate the results.
    let mut sorted = quick_sort(left);
    sorted.extend(middle);
    sorted.extend(quick_sort(right));
    sorted
}

fn main() {
    // Read the entire input.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();
    let mut lines = input.lines();

    // Discard the first line (mirrors the first `input()` call in Python).
    lines.next();

    // Parse the second line into a vector of integers.
    let numbers_line = lines.next().unwrap_or("");
    let numbers: Vec<i32> = numbers_line
        .split_whitespace()
        .map(|s| s.parse().expect("invalid integer"))
        .collect();

    // Sort the numbers and count distinct elements.
    let sorted_numbers = quick_sort(numbers);
    let unique_values: HashSet<i32> = sorted_numbers.into_iter().collect();

    // Output the count of distinct sorted numbers.
    println!("{}", unique_values.len());
}
