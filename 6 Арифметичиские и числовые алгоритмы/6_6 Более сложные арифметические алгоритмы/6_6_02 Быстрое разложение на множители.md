Входные данные
Дано натуральное число N, 2 ≤ N ≤ 1018. Известно, что оно является произведением двух простых чисел.

Выходные данные
Выведите простые делители данного числа в порядке неубывания.

Sample Input:
15

Sample Output:
3 5

Напишите программу. Тестируется через stdin → stdout

def foo(x)
  if x.even?
    [2, x / 2]
  else
    d1 = Math.sqrt(x).to_i
    d1 += 1 if d1.even?
    while d1 > 2
      return [d1, x / d1] if (x % d1).zero?

      d1 -= 2
    end
  end
end

# rust
use std::io;

fn mul_mod(mut a: u128, mut b: u128, m: u128) -> u128 {
    (a as u128 * b as u128 % m as u128) as u128
}

fn power(mut a: u128, mut b: u128, m: u128) -> u128 {
    let mut res = 1;
    a %= m;
    while b > 0 {
        if b % 2 == 1 { res = mul_mod(res, a, m); }
        a = mul_mod(a, a, m);
        b /= 2;
    }
    res
}

fn is_prime(n: u128) -> bool {
    if n < 2 { return false; }
    if n == 2 || n == 3 { return true; }
    if n % 2 == 0 { return false; }
    let mut d = n - 1;
    let mut s = 0;
    while d % 2 == 0 {
        d /= 2;
        s += 1;
    }
    // Базы для Miller-Rabin теста (достаточно для n < 3 * 10^18)
    let bases = [2, 3, 5, 7, 11, 13, 17, 19, 23];
    for &a in &bases {
        if n <= a { break; }
        let mut x = power(a, d, n);
        if x == 1 || x == n - 1 { continue; }
        let mut composite = true;
        for _ in 0..s - 1 {
            x = mul_mod(x, x, n);
            if x == n - 1 {
                composite = false;
                break;
            }
        }
        if composite { return false; }
    }
    true
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        a %= b;
        std::mem::swap(&mut a, &mut b);
    }
    a
}

fn pollard_rho(n: u128) -> u128 {
    if n % 2 == 0 { return 2; }
    if is_prime(n) { return n; }
    let mut x = 2;
    let mut y = 2;
    let mut d = 1;
    let mut c = 1;
    let f = |x: u128, c: u128, m: u128| (mul_mod(x, x, m) + c) % m;

    while d == 1 {
        x = f(x, c, n);
        y = f(f(y, c, n), c, n);
        d = gcd(if x > y { x - y } else { y - x }, n);
        if d == n {
            // Если зациклились, пробуем с другими параметрами
            x = 2; y = 2; c += 1; d = 1;
        }
    }
    d
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let n: u128 = input.trim().parse().unwrap();

    let d = pollard_rho(n);
    let d2 = n / d;

    if d < d2 {
        println!("{} {}", d, d2);
    } else {
        println!("{} {}", d2, d);
    }
}
