# python
from math import gcd

n = int(input())
f = set()

for p in range(1, n):
    q = n
    while q > 1 and p / q < 1:
        d = gcd(p, q)
        f.add((p / q, p // d, q // d))
        q -= 1
        
for _, p, q in sorted(f):
     print(f'{p}/{q}')

# rust
use std::collections::HashSet;
use std::io::{self, Read};

/// Compute the greatest common divisor using Euclid's algorithm.
fn gcd(mut a: usize, mut b: usize) -> usize {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn main() {
    // ----- read input -----
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).expect("failed to read stdin");
    let n: usize = input.trim().parse().expect("invalid integer");

    // ----- generate reduced fractions -----
    let mut fraction_set: HashSet<(usize, usize)> = HashSet::new();

    for numerator in 1..n {
        let mut denominator = n;
        while denominator > 1 && (numerator as f64) / (denominator as f64) < 1.0 {
            let d = gcd(numerator, denominator);
            let reduced_num = numerator / d;
            let reduced_den = denominator / d;
            fraction_set.insert((reduced_num, reduced_den));
            denominator -= 1;
        }
    }

    // ----- sort by numeric value -----
    let mut fractions: Vec<(usize, usize)> = fraction_set.into_iter().collect();
    fractions.sort_by(|a, b| {
        // Compare a.num / a.den  with  b.num / b.den  via cross‑multiplication
        (a.0 as u128 * b.1 as u128).cmp(&(b.0 as u128 * a.1 as u128))
    });

    // ----- output -----
    for (num, den) in fractions {
        println!("{}/{}", num, den);
    }
}
