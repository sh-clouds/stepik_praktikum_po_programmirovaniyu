use std::io;

// Проверка: является ли строка "0"
fn is_zero(s: &str) -> bool {
    s.chars().all(|c| c == '0') || s.is_empty()
}

// Сравнение двух больших чисел в виде строк
fn is_less(a: &str, b: &str) -> bool {
    if a.len() != b.len() {
        return a.len() < b.len();
    }
    a < b
}

// Вычитание: a - b (предполагаем a >= b)
fn subtract(a: &str, b: &str) -> String {
    let mut res = Vec::new();
    let a_bytes = a.as_bytes();
    let b_bytes = b.as_bytes();
    let mut borrow = 0;

    let mut i = a_bytes.len() as i32 - 1;
    let mut j = b_bytes.len() as i32 - 1;

    while i >= 0 {
        let mut digit_a = (a_bytes[i as usize] - b'0') as i32 - borrow;
        let digit_b = if j >= 0 { (b_bytes[j as usize] - b'0') as i32 } else { 0 };

        if digit_a < digit_b {
            digit_a += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }

        res.push(((digit_a - digit_b) as u8 + b'0') as char);
        i -= 1;
        j -= 1;
    }

    let s: String = res.into_iter().rev().collect();
    let trimmed = s.trim_start_matches('0');
    if trimmed.is_empty() { "0".to_string() } else { trimmed.to_string() }
}

// Упрощенный поиск остатка (a % b) через вычитание для GCD
fn big_mod(mut a: String, b: String) -> String {
    if is_zero(&b) { return a; }
    while !is_less(&a, &b) {
        // Для ускорения: можно было бы вычитать b, умноженное на степень 10,
        // но для учебной задачи оставим базовое вычитание.
        a = subtract(&a, &b);
    }
    a
}

fn gcd(mut a: String, mut b: String) -> String {
    while !is_zero(&b) {
        let temp = big_mod(a, b.clone());
        a = b;
        b = temp;
    }
    a
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let parts: Vec<&str> = input.split_whitespace().collect();

    if parts.len() < 2 { return; }

}
