МегаНОД

Дано N чисел. Найти самое большое число, на которое делятся все N чисел.

Входные данные
В первой строке дано число N. Во второй строке даны через пробел N чисел (1 <= N <= 1000).

Выходные данные
Выведите искомое число.

# rust
use std::io::{self, Read};

/// Compute the greatest common divisor (GCD) of two integers using the
/// Euclidean algorithm.
fn gcd(mut a: i64, mut b: i64) -> i64 {
    while b != 0 {
        let remainder = a % b;
        a = b;
        b = remainder;
    }
    a.abs()
}

fn main() {
    // Read the whole input.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).unwrap();

    // Split by whitespace; the first token is ignored.
    let mut tokens = input.split_whitespace();
    tokens.next(); // discard first line / token

    // Parse the remaining tokens as integers.
    let numbers: Vec<i64> = tokens.map(|t| t.parse::<i64>().unwrap()).collect();

    // Calculate the GCD of all numbers. If there are no numbers, the result is 0.
    let result = numbers.iter().fold(0, |acc, &value| {
        if acc == 0 {
            value
        } else {
            gcd(acc, value)
        }
    });

    // Output the result.
    println!("{}", result);
}
