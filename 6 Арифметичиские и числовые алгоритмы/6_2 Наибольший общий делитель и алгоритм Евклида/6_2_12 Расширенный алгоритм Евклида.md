Входные данные - натуральные числа и не превышают по модулю 10000.

Выходные данные
Выведите ответ на задачу.

def extended_gcd(a, b):
    if not b:
        return a, 1, 0
    else:
        gcd, x, y = extended_gcd(b, a % b)
        return gcd, y, x - (a // b) * y


a, b, c = map(int, input().split())
gcd, x0, y0 = extended_gcd(a, b)
if c % gcd:
    print("Impossible")
else:
    print(gcd, x0 * (c // gcd), y0 * (c // gcd))

# rust
use std::io::{self, Read};

/// Computes the extended Euclidean algorithm.
/// Returns a tuple (gcd, x, y) such that a*x + b*y = gcd.
fn extended_gcd(a: i64, b: i64) -> (i64, i64, i64) {
    if b == 0 {
        (a, 1, 0)
    } else {
        let (gcd, x1, y1) = extended_gcd(b, a % b);
        (gcd, y1, x1 - (a / b) * y1)
    }
}

fn main() {
    // Read the whole input.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).expect("Failed to read stdin");

    // Expect three integers separated by whitespace.
    let mut nums = input
        .split_whitespace()
        .map(|s| s.parse::<i64>().expect("Invalid integer"));

    let a = nums.next().expect("Missing first integer");
    let b = nums.next().expect("Missing second integer");
    let c = nums.next().expect("Missing third integer");

    let (gcd, x0, y0) = extended_gcd(a, b);

    if c % gcd != 0 {
        println!("Impossible");
    } else {
        let factor = c / gcd;
        let x = x0 * factor;
        let y = y0 * factor;
        println!("{} {} {}", gcd, x, y);
    }
}
