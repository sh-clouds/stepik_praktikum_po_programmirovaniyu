Частное от деления на длинное

Напишите программу, вычисляющую частное от деления двух целых неотрицательных чисел.

Входные данные
Дано два числа, длина каждого не превосходит 250 знаков, каждое число вводится на отдельной строке.

Выходные данные
Выведите частное от деления первого на второе.

Sample Input:
7
3

Sample Output:
2

Напишите программу. Тестируется через stdin → stdout

use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Считываем два числа как строки
    let s1 = lines.next().unwrap().unwrap();
    let s2 = lines.next().unwrap().unwrap();

    let num1 = s1.trim();
    let num2 = s2.trim();

    // Обработка деления на ноль (если подразумевается)
    if num2 == "0" {
        return; 
    }

    println!("{}", big_int_div(num1, num2));
}

fn big_int_div(a: &str, b: &str) -> String {
    let mut quotient = String::new();
    let mut current_remainder = String::new();

    for c in a.chars() {
        current_remainder.push(c);
        // Убираем ведущие нули
        while current_remainder.len() > 1 && current_remainder.starts_with('0') {
            current_remainder.remove(0);
        }

        let mut count = 0;
        // Пока остаток >= b, вычитаем b
        while is_greater_or_equal(&current_remainder, b) {
            current_remainder = subtract(&current_remainder, b);
            count += 1;
        }
        quotient.push(std::char::from_digit(count, 10).unwrap());
    }

    // Убираем ведущие нули в ответе
    let result = quotient.trim_start_matches('0');
    if result.is_empty() {
        "0".to_string()
    } else {
        result.to_string()
    }
}

// Функция сравнения длинных чисел в виде строк
fn is_greater_or_equal(a: &str, b: &str) -> bool {
    if a.len() != b.len() {
        return a.len() > b.len();
    }
    a >= b
}

// Функция вычитания длинных чисел: a - b
fn subtract(a: &str, b: &str) -> String {
    let mut result = String::new();
    let a_bytes = a.as_bytes();
    let b_bytes = b.as_bytes();
    let mut i = a_bytes.len() as i32 - 1;
    let mut j = b_bytes.len() as i32 - 1;
    let mut borrow = 0;

    while i >= 0 {
        let digit_a = (a_bytes[i as usize] - b'0') as i32;
        let digit_b = if j >= 0 { (b_bytes[j as usize] - b'0') as i32 } else { 0 };

        let mut sub = digit_a - digit_b - borrow;
        if sub < 0 {
            sub += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push((sub as u8 + b'0') as char);
        i -= 1;
        j -= 1;
    }

    let res = result.chars().rev().collect::<String>();
    let trimmed = res.trim_start_matches('0');
    if trimmed.is_empty() { "0".to_string() } else { trimmed.to_string() }
}
