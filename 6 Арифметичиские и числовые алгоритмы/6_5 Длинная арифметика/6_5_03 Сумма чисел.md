let transform = (n1, n2, res_len) => {
    n1.push(...new Array(res_len).fill(0));
    n1 = n1.slice(0, res_len);
    
    n2.push(...new Array(res_len).fill(0));
    n2 = n2.slice(0, res_len);
    return [n1, n2];
};

let sum = (n1, n2) => {
    res_len = Math.max(n1.length, n2.length) + 1;
    [n1, n2] = transform(n1, n2, res_len)
    
    let res = [];
    let to_mind = 0;
    for (let i = 0; i < res_len; i++) {
        let tmp = n1[i] + n2[i] + to_mind;
        res.push(tmp % 10);
        to_mind = Math.floor(tmp / 10);
    };
    return res.reverse().join('').replace(/^0+/, '');
};

# rust
use std::io;

fn main() {
    let mut num1 = String::new();
    let mut num2 = String::new();

    // Чтение двух строк
    io::stdin().read_line(&mut num1).unwrap();
    io::stdin().read_line(&mut num2).unwrap();

    let s1 = num1.trim();
    let s2 = num2.trim();

    // Преобразуем строки в векторы байт-цифр и переворачиваем для сложения с младших разрядов
    let mut v1: Vec<u8> = s1.chars().map(|c| c.to_digit(10).unwrap() as u8).rev().collect();
    let mut v2: Vec<u8> = s2.chars().map(|c| c.to_digit(10).unwrap() as u8).rev().collect();

    let max_len = std::cmp::max(v1.len(), v2.len());
    let mut result = Vec::with_capacity(max_len + 1);
    let mut carry = 0; // Перенос

    for i in 0..max_len {
        let digit1 = if i < v1.len() { v1[i] } else { 0 };
        let digit2 = if i < v2.len() { v2[i] } else { 0 };

        let sum = digit1 + digit2 + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }

    if carry > 0 {
        result.push(carry);
    }

    // Переворачиваем обратно и выводим
    result.reverse();
    for digit in result {
        print!("{}", digit);
    }
    println!();
}
