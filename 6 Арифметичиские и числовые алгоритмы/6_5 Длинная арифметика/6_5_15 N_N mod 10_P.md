N^N mod 10^P

Входные данные
Даны два числа N и P (1≤N≤109, 1≤P≤100).

Выходные данные
Вычислите N**N mod 10**P.

Sample Input:
5 3

Sample Output:
125

Напишите программу. Тестируется через stdin → stdout

# rust
use std::io;

#[derive(Clone)]
struct BigMod {
    digits: Vec<u32>, // Храним по одной цифре для простоты (индекс 0 - единицы)
    p: usize,
}

impl BigMod {
    // Создание из обычного числа N
    fn from_u64(mut n: u64, p: usize) -> Self {
        let mut digits = Vec::new();
        while n > 0 {
            digits.push((n % 10) as u32);
            n /= 10;
        }
        digits.truncate(p);
        BigMod { digits, p }
    }

    // Умножение (self * other) mod 10^P
    fn mul(&self, other: &Self) -> Self {
        let mut res_digits = vec![0u32; self.p];
        
        for i in 0..self.digits.len() {
            if i >= self.p { break; }
            let mut carry = 0u64;
            for j in 0..other.digits.len() {
                if i + j >= self.p { break; }
                
                let prod = self.digits[i] as u64 * other.digits[j] as u64 
                           + res_digits[i + j] as u64 + carry;
                
                res_digits[i + j] = (prod % 10) as u32;
                carry = prod / 10;
            }
            
            // Распространяем перенос, пока он есть и мы в пределах P
            let mut k = i + other.digits.len();
            while carry > 0 && k < self.p {
                let sum = res_digits[k] as u64 + carry;
                res_digits[k] = (sum % 10) as u32;
                carry = sum / 10;
                k += 1;
            }
        }
        
        BigMod { digits: res_digits, p: self.p }
    }

    fn to_string(&self) -> String {
        if self.digits.is_empty() { return "0".to_string(); }
        let mut s: String = self.digits.iter().rev()
            .map(|&d| std::char::from_digit(d, 10).unwrap())
            .collect();
        
        // Убираем ведущие нули, но оставляем один, если число 0
        let trimmed = s.trim_start_matches('0');
        if trimmed.is_empty() { "0".to_string() } else { trimmed.to_string() }
    }
}

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let nums: Vec<u64> = input.split_whitespace()
        .map(|s| s.parse().expect("Parse error"))
        .collect();

    let n = nums[0];
    let p = nums[1] as usize;

    // Быстрое возведение в степень: n^n mod 10^p
    let mut res = BigMod::from_u64(1, p);
    let mut base = BigMod::from_u64(n, p);
    let mut exp = n;

    while exp > 0 {
        if exp % 2 == 1 {
            res = res.mul(&base);
        }
        base = base.mul(&base);
        exp /= 2;
    }

    println!("{}", res.to_string());
}
