Разность чисел

Напишите программу, вычисляющую разность двух целых неотрицательных чисел. Гарантируется, что первое число не меньше второго.

Входные данные
Даны два числа, второе из них не превышает первого. Числа не превышают 10100

Выходные данные
Выведите разность данных чисел.

Sample Input:
7
5

Sample Output:
2

Напишите программу. Тестируется через stdin → stdout

# java
import java.util.Scanner;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        BigInteger a = new BigInteger(sc.next());
        BigInteger b = new BigInteger(sc.next());
        
        System.out.println(a.subtract(b));
    }
}



# rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    // Считываем два числа как строки
    let s1 = lines.next().unwrap().unwrap();
    let s2 = lines.next().unwrap().unwrap();

    let n1: Vec<u8> = s1.trim().bytes().map(|b| b - b'0').rev().collect();
    let n2: Vec<u8> = s2.trim().bytes().map(|b| b - b'0').rev().collect();

    let mut result = Vec::new();
    let mut borrow = 0;

    // Поразрядное вычитание
    for i in 0..n1.len() {
        let digit1 = n1[i];
        let digit2 = if i < n2.len() { n2[i] } else { 0 };

        let mut sub = digit1 as i16 - digit2 as i16 - borrow;
        if sub < 0 {
            sub += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push((sub as u8) + b'0');
    }

    // Убираем ведущие нули
    while result.len() > 1 && *result.last().unwrap() == b'0' {
        result.pop();
    }

    // Выводим результат
    println!("{}", String::from_utf8(result.into_iter().rev().collect()).unwrap());
}
