Сравнение чисел

Напишите программу, которая сравнит два целых неотрицательных числа и выведет результат сравнения

Входные данные
Даны два целых неотрицательных числа, в записи каждого не более 200 цифр.

Выходные данные
Выведите знак <, если первое число меньше второго, знак >, если первое больше второго или знак =, если числа совпадают.

Sample Input:
5556
5554

Sample Output:
>

Напишите программу. Тестируется через stdin → stdout

# java
import java.util.Scanner;
import java.math.BigInteger;

class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        
        BigInteger a = new BigInteger(sc.next());
        BigInteger b = new BigInteger(sc.next());
        
        int result = a.compareTo(b);
        
        System.out.println(result == 0 ? '=' : result == 1 ? '>' : '<');
    }
}


# C
#include <stdio.h>
#include <string.h>
#define SIZE 201

int get_correct_num(char *s) {
    fgets(s, SIZE, stdin);
    int len = strlen(s);
    char *e = s + len - 1;
    if(*e == '\n')
        *e = '\0', len--;
    return len;
}

int main() {
    char num1[SIZE], num2[SIZE];
    int len1 = get_correct_num(num1);   
    int len2 = get_correct_num(num2);
    
    if(len1 < len2)
        return puts("<") && 0;
    if(len1 > len2)
        return puts(">") && 0;
    
    char *p1 = num1, *p2 = num2;
    while(*p1 && *p1 == *p2)
        p1++, p2++;
    
    puts(*p1 < *p2 ? "<": 
         *p1 > *p2 ? ">": "=");
    return 0;
}



# rust
use std::io;

fn main() {
    let mut input1 = String::new();
    let mut input2 = String::new();

    io::stdin().read_line(&mut input1).unwrap();
    io::stdin().read_line(&mut input2).unwrap();

    // Удаляем ведущие нули и лишние пробелы/символы перевода строки
    let s1 = trim_leading_zeros(input1.trim());
    let s2 = trim_leading_zeros(input2.trim());

    // 1. Сравнение по длине
    if s1.len() < s2.len() {
        println!("<");
    } else if s1.len() > s2.len() {
        println!(">");
    } else {
        // 2. Длины равны, сравниваем лексикографически
        if s1 < s2 {
            println!("<");
        } else if s1 > s2 {
            println!(">");
        } else {
            println!("=");
        }
    }
}

// Функция для удаления ведущих нулей
fn trim_leading_zeros(s: &str) -> &str {
    let trimmed = s.trim_start_matches('0');
    if trimmed.is_empty() {
        "0" // Если число было "000", возвращаем "0"
    } else {
        trimmed
    }
}
