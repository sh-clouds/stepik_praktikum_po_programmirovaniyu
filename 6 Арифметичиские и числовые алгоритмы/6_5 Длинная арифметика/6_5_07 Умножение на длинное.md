Напишите программу, которая вычисляет произведение двух целых неотрицательных чисел.

Входные данные
Даны два числа, каждое в отдельной строке, длина каждого не превышает 250 знаков.

Выходные данные
Выведите произведение данных чисел.

Sample Input:
2
9

Sample Output:
18

Напишите программу. Тестируется через stdin → stdout

# rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let lines: Vec<String> = stdin.lock().lines().map(|l| l.unwrap().trim().to_string()).collect();
    
    if lines.len() < 2 { return; }
    
    let res = multiply(&lines[0], &lines[1]);
    println!("{}", res);
}

fn multiply(s1: &str, s2: &str) -> String {
    if s1 == "0" || s2 == "0" { return "0".to_string(); }

    let n1: Vec<u32> = s1.chars().map(|c| c.to_digit(10).unwrap()).collect();
    let n2: Vec<u32> = s2.chars().map(|c| c.to_digit(10).unwrap()).collect();
    let mut result = vec![0; n1.len() + n2.len()];

    for i in (0..n1.len()).rev() {
        for j in (0..n2.len()).rev() {
            let mult = n1[i] * n2[j];
            let sum = mult + result[i + j + 1];
            
            result[i + j + 1] = sum % 10;
            result[i + j] += sum / 10;
        }
    }

    // Убираем ведущие нули и превращаем в строку
    let mut start = 0;
    while start < result.len() - 1 && result[start] == 0 {
        start += 1;
    }
    
    result[start..].iter().map(|&d| d.to_string()).collect()
}
