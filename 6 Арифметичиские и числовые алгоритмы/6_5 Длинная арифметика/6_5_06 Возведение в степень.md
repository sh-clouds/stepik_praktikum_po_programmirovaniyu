# rust
use std::io;

fn main() {
    let mut input = String::new();
    io::stdin().read_line(&mut input).unwrap();
    let n: usize = input.trim().parse().unwrap();

    if n == 0 {
        println!("1");
        return;
    }

    let mut digits = vec![1u8]; // Храним число поразрядно

    for _ in 0..n {
        let mut carry = 0u32;
        for digit in digits.iter_mut() {
            let prod = (*digit as u32) * 2 + carry;
            *digit = (prod % 10) as u8;
            carry = prod / 10;
        }
        while carry > 0 {
            digits.push((carry % 10) as u8);
            carry /= 10;
        }
    }

    // Вывод в обратном порядке
    for digit in digits.iter().rev() {
        print!("{}", digit);
    }
    println!();
}
