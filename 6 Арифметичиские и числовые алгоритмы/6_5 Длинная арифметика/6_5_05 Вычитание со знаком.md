Вычитание со знаком

Напишите программу, которая находит разность двух целых неотрицательных чисел. Длина каждого числа не превышает 250 знаков.

Входные данные
Даны два числа, каждое записано в отдельной строке

Выходные данные
Выведите разность чисел (учтите, что может получиться отрицательный результат).

Sample Input:
5
7

Sample Output:
-2

Напишите программу. Тестируется через stdin → stdout

# rust
use std::io;

fn main() {
    let mut s1 = String::new();
    let mut s2 = String::new();

    io::stdin().read_line(&mut s1).unwrap();
    io::stdin().read_line(&mut s2).unwrap();

    let s1 = s1.trim();
    let s2 = s2.trim();

    if compare(s1, s2) == -1 {
        // Если s1 < s2, считаем s2 - s1 и добавляем минус
        print!("-");
        println!("{}", subtract(s2, s1));
    } else {
        println!("{}", subtract(s1, s2));
    }
}

// Сравнение двух строк-чисел (1, если a>b, -1 если a<b, 0 если a=b)
fn compare(a: &str, b: &str) -> i8 {
    let a = a.trim_start_matches('0');
    let b = b.trim_start_matches('0');
    if a.len() > b.len() { return 1; }
    if a.len() < b.len() { return -1; }
    if a > b { return 1; }
    if a < b { return -1; }
    0
}

// Вычитание длинных чисел: a - b (гарантируется a >= b)
fn subtract(a: &str, b: &str) -> String {
    let a_bytes = a.as_bytes();
    let b_bytes = b.as_bytes();
    let mut result = Vec::new();
    let mut borrow = 0;

    let len_a = a_bytes.len();
    let len_b = b_bytes.len();

    for i in 0..len_a {
        let digit_a = (a_bytes[len_a - 1 - i] - b'0') as i32;
        let digit_b = if i < len_b {
            (b_bytes[len_b - 1 - i] - b'0') as i32
        } else {
            0
        };

        let mut sub = digit_a - digit_b - borrow;
        if sub < 0 {
            sub += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        result.push((sub as u8) + b'0');
    }

    // Удаляем ведущие нули
    while result.len() > 1 && *result.last().unwrap() == b'0' {
        result.pop();
    }

    result.reverse();
    String::from_utf8(result).unwrap()
}
