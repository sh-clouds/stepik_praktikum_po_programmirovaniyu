Остаток от деления на длинное

Напишите программу, которая вычисляет остаток от деления двух целых неотрицательных чисел.

Входные данные
Дано два числа, длина каждого не превосходит 250 знаков, каждое число вводится на отдельной строке.

Выходные данные
Выведите остаток от деления первого на второе.

Sample Input:
7
3

Sample Output:
1

Напишите программу. Тестируется через stdin → stdout

# rust
use std::io::{self, BufRead};

fn main() {
    let stdin = io::stdin();
    let mut lines = stdin.lock().lines();

    let a_str = lines.next().unwrap().unwrap().trim().to_string();
    let b_str = lines.next().unwrap().unwrap().trim().to_string();

    if b_str == "0" {
        return; // Деление на ноль не определено
    }

    println!("{}", get_remainder(&a_str, &b_str));
}

fn get_remainder(a: &str, b: &str) -> String {
    let mut remainder = String::from("0");

    for digit in a.chars() {
        // Добавляем следующую цифру к текущему остатку
        if remainder == "0" {
            remainder = digit.to_string();
        } else {
            remainder.push(digit);
        }

        // Вычитаем делитель b из remainder, пока remainder >= b
        while is_ge(&remainder, b) {
            remainder = subtract(&remainder, b);
        }
    }

    remainder
}

// Сравнение двух строк-чисел (больше или равно)
fn is_ge(a: &str, b: &str) -> bool {
    if a.len() != b.len() {
        return a.len() > b.len();
    }
    a >= b
}

// Вычитание двух строк-чисел (a - b), где a >= b
fn subtract(a: &str, b: &str) -> String {
    let a_bytes = a.as_bytes();
    let b_bytes = b.as_bytes();
    let mut result = Vec::new();
    let mut borrow = 0i32;

    let mut i = a_bytes.len() as i32 - 1;
    let mut j = b_bytes.len() as i32 - 1;

    while i >= 0 {
        let digit_a = (a_bytes[i as usize] - b'0') as i32;
        let digit_b = if j >= 0 { (b_bytes[j as usize] - b'0') as i32 } else { 0 };

        let mut diff = digit_a - digit_b - borrow;
        if diff < 0 {
            diff += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }

        result.push((diff as u8 + b'0') as char);
        i -= 1;
        j -= 1;
    }

    let mut s: String = result.into_iter().rev().collect();
    let trimmed = s.trim_start_matches('0');
    if trimmed.is_empty() { "0".to_string() } else { trimmed.to_string() }
}
