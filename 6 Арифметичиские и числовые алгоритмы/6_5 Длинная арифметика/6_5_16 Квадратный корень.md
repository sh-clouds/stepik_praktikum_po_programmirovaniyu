# ruby
def largest_square_root(input_value)
  # Преобразуем A в целое число
  integer_value = input_value.to_i
  
  # Определяем границы для бинарного поиска
  left_boundary, right_boundary = 0, integer_value
  
  while left_boundary <= right_boundary
    mid_point = (left_boundary + right_boundary) / 2
    if mid_point * mid_point <= integer_value
      left_boundary = mid_point + 1
    else
      right_boundary = mid_point - 1
    end
  end
  
  return right_boundary
end

    let target = parse(s);
    
    let mut low = vec![0];
    let mut high = target.clone();
    let mut res = vec![0];

    while compare(&low, &high) <= 0 {
        let mid = add(&low, &high);
        let mid = div2(&mid);
        
        let s_mid = mul(&mid, &mid);

        }

// Вспомогательные функции для работы с вектором цифр (BigInt)
fn parse(s: &str) -> Vec<u8> {
    s.bytes().map(|b| b - b'0').collect()
}

fn to_str(v: &[u8]) -> String {
    if v.is_empty() { return "0".to_string(); }
    v.iter().map(|&d| (d + b'0') as char).collect()
}

fn compare(a: &[u8], b: &[u8]) -> i8 {
    if a.len() != b.len() {
        return if a.len() > b.len() { 1 } else { -1 };
    }
    for (&x, &y) in a.iter().zip(b.iter()) {
        if x != y { return if x > y { 1 } else { -1 }; }
    }
    0
}

fn add(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut res = Vec::new();
    let (mut i, mut j, mut carry) = (a.len() as i32 - 1, b.len() as i32 - 1, 0u8);
    while i >= 0 || j >= 0 || carry > 0 {
        let sum = carry + if i >= 0 { a[i as usize] } else { 0 } + if j >= 0 { b[j as usize] } else { 0 };
        res.push(sum % 10);
        carry = sum / 10;
        i -= 1; j -= 1;
    }
    res.reverse();
    res
}

fn sub(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut res = Vec::new();
    let (mut i, mut j, mut borrow) = (a.len() as i32 - 1, b.len() as i32 - 1, 0i8);
    while i >= 0 {
        let mut diff = a[i as usize] as i8 - borrow - if j >= 0 { b[j as usize] as i8 } else { 0 };
        if diff < 0 { diff += 10; borrow = 1; } else { borrow = 0; }
        res.push(diff as u8);
        i -= 1; j -= 1;
    }
    while res.len() > 1 && res.last() == Some(&0) { res.pop(); }
    res.reverse();
    res
}

fn mul(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut res = vec![0u8; a.len() + b.len()];
    for i in (0..a.len()).rev() {
        let mut carry = 0u8;
        for j in (0..b.len()).rev() {
            let cur = res[i + j + 1] + a[i] * b[j] + carry;
            res[i + j + 1] = cur % 10;
            carry = cur / 10;
        }
        res[i] += carry;
    }
    let start = res.iter().position(|&x| x != 0).unwrap_or(res.len() - 1);
    res[start..].to_vec()
}

fn div2(a: &[u8]) -> Vec<u8> {
    let mut res = Vec::new();
    let mut rem = 0u16;
    for &d in a {
        let cur = rem * 10 + d as u16;
        res.push((cur / 2) as u8);
        rem = cur % 2;
    }
    let start = res.iter().position(|&x| x != 0).unwrap_or(res.len() - 1);
    res[start..].to_vec()
}
