Инопланетянин

Во время эксперимента Накодиллы было случайно получено сообщение инопланетян, содержащее формулу вида A + B = C.

Общественности стало интересно, какую же систему счисления используют инопланетяне. Так как внеземная цивилизация была достаточно развита, чтобы отправить межпланетное сообщение, Накодилла предположил, что основание системы счисления довольно мало. Требуется написать программу, которая находит минимальное основание системы счисления, при котором данное равенство выполняется.

Входные данные
В единственной строке входных данных содержится равенство вида A + B = C. Строка не содержит пробелы, числа состоят из цифр от 0 до 9 и заглавных латинских букв от А до Z

Выходные данные
Требуется вывести единственное число – искомое основание системы счисления. Если такой системы счисления не существует, то вывести -1. Гарантируется, что ответ не превышает 36.

Sample Input:
2+2=4

Sample Output:
5

Напишите программу. Тестируется через stdin → stdout

# ruby
digit_characters = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ'
user_input = gets.chomp
user_input = user_input.gsub('+', ' ').gsub('=', ' ').upcase
min_number_system = digit_characters.index(user_input.chars.max) + 1
first_number, second_number, result_number = user_input.split
found_base = -1

(min_number_system..36).each do |base|
  if first_number.to_i(base) + second_number.to_i(base) == result_number.to_i(base)
    found_base = base
    break
  end
end

puts found_base

digits = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ'

def to_ten_base(number: str, from_base: int) -> int:
    return sum(digits.find(symbol) * from_base ** i
               for i, symbol in enumerate(number[::-1]))

*AB, C = input().split('=')
A, B = AB[0].split('+')
del AB
start_base = digits.find(max(A + B + C)) + 1

for base in range(start_base, 37):
    a = to_ten_base(A, base)
    b = to_ten_base(B, base)
    c = to_ten_base(C, base)
    if a + b == c:
        print(base)
        break
else:
    print(-1)

# rust
use std::io::{self, Read};

/// Returns the numeric value of a digit character according to
/// the custom digit table "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ".
fn digit_value(c: char) -> Option<u32> {
    const DIGITS: &str = "0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ";
    DIGITS.chars().position(|d| d == c).map(|i| i as u32)
}

/// Parses a number written in a given base (2 .. 36). Returns `None` if the
/// string contains an invalid digit for that base or the conversion overflows.
fn parse_in_base(s: &str, base: u32) -> Option<i64> {
    // `i64::from_str_radix` expects the base to be within 2..=36.
    if base < 2 || base > 36 {
        return None;
    }
    i64::from_str_radix(s, base).ok()
}

fn main() {
    // ----- read whole stdin -------------------------------------------------
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).expect("failed to read stdin");
    let mut user_input = input.trim_end().to_string();

    // ----- preprocess the string (replace '+' and '=', uppercase) ----------
    user_input = user_input
        .replace('+', " ")
        .replace('=', " ")
        .to_uppercase();

    // ----- split into three parts -------------------------------------------
    let parts: Vec<&str> = user_input.split_whitespace().collect();
    if parts.len() != 3 {
        eprintln!("expected exactly three tokens");
        println!("-1");
        return;
    }
    let first_number = parts[0];
    let second_number = parts[1];
    let result_number = parts[2];

    // ----- compute minimal possible base ------------------------------------
    // Find the maximal digit character present in the whole input.
    let max_char = user_input
        .chars()
        .filter(|c| c.is_ascii_alphanumeric())
        .max()
        .unwrap(); // there is at least one alphanumeric char

    let min_base = match digit_value(max_char) {
        Some(v) => (v + 1) as u32,
        None => {
            eprintln!("invalid character found");
            println!("-1");
            return;
        }
    };

    // ----- search for a base that satisfies the equation --------------------
    let mut found_base: i32 = -1;
    for base in min_base..=36 {
        if let (Some(a), Some(b), Some(c)) = (
            parse_in_base(first_number, base),
            parse_in_base(second_number, base),
            parse_in_base(result_number, base),
        ) {
            if a + b == c {
                found_base = base as i32;
                break;
            }
        }
    }

    // ----- output ------------------------------------------------------------
    println!("{}", found_base);
}
