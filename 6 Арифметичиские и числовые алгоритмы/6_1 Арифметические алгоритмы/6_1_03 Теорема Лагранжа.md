Теорема Лагранжа

Теорема Лагранжа утверждает, что любое натуральное число можно представить в виде суммы четырех точных квадратов. По данному числу n найдите такое представление: напечатайте от 1 до 4 натуральных чисел, квадраты которых дают в сумме данное число.

Входные данные
Программа получает на вход одно натуральное число n < 10000.

Выходные данные
Программа должна вывести от 1 до 4 натуральных чисел, квадраты которых дают в сумме данное число.

Sample Input 1:
3

Sample Output 1:
1 1 1

Sample Input 2:
7

Sample Output 2:
2 1 1 1

# ruby
def lagrange(x, pow, q)
    b = (x**(1.to_f/pow)).to_i
    if (x - b ** pow) == 0 then return b.to_s end    
    if q == 1 then return 0 end    

    while (lagrange(x - b ** pow, pow, q - 1) == 0)
        b -= 1
        if b <= 0 then return 0 end
    end
        
    return b.to_s + ' ' + lagrange(x - b ** pow, pow, q - 1)   
end

def lagrang(n):
    res_dict = {i ** 2 + j ** 2: [i, j] for i in range(int(n ** 0.5) + 1) for j in range(i, int(n ** 0.5) + 1)}
    for i in res_dict:
        if n - i in res_dict:
            return list(filter(lambda x: x, res_dict[i] + res_dict[n - i]))

# rust
use std::io::{self, Read};

/// Calculates the bounds used for the decreasing ranges.
///
/// The function mimics:
/// - If no arguments are given it treats the first argument as `0`.
/// - It returns a tuple `(limit, -1, -1)` where `limit` is the integer part
///   of the square‑root of `first - Σ(i²)` for the remaining non‑zero arguments.
fn bounds(args: &[i64]) -> (i64, i64, i64) {
    let first = if args.is_empty() { 0 } else { args[0] };
    let sum_sq: i64 = args
        .iter()
        .skip(1)                // skip the first element
        .filter(|&&x| x != 0)   // keep only truthy (non‑zero) values
        .map(|&x| x * x)
        .sum();

    let diff = first - sum_sq;
    // `diff` is always non‑negative in the original algorithm,
    // but we guard against negative values to avoid NaN.
    let limit = if diff > 0 {
        (diff as f64).sqrt() as i64
    } else {
        0
    };
    (limit, -1, -1)
}

/// Searches for four non‑negative integers `a, b, c, d` such that
/// `a² + b² + c² + d² == n`. The search order follows the
/// implementation: each variable iterates from its computed bound
/// down to `0`.
fn lagrange(n: i64) -> Option<(i64, i64, i64, i64)> {
    // a‑loop
    let (limit_a, _, _) = bounds(&[n]);
    for a in (0..=limit_a).rev() {
        // b‑loop
        let (limit_b, _, _) = bounds(&[n, a]);
        for b in (0..=limit_b).rev() {
            // c‑loop
            let (limit_c, _, _) = bounds(&[n, a, b]);
            for c in (0..=limit_c).rev() {
                // d‑loop
                let (limit_d, _, _) = bounds(&[n, a, b, c]);
                for d in (0..=limit_d).rev() {
                    if a * a + b * b + c * c + d * d == n {
                        return Some((a, b, c, d));
                    }
                }
            }
        }
    }
    None
}

fn main() {
    // Read the whole input, trim, and parse the integer N.
    let mut input = String::new();
    io::stdin().read_to_string(&mut input).expect("Failed to read stdin");
    let n: i64 = input.trim().parse().expect("Input is not a valid integer");

    if let Some((a, b, c, d)) = lagrange(n) {
        // Collect only the non‑zero components, convert them to strings,
        // and print them space‑separated.
        let result = [a, b, c, d]
            .iter()
            .filter(|&&x| x != 0)
            .map(|x| x.to_string())
            .collect::<Vec<_>>()
            .join(" ");
        println!("{}", result);
    } else {
        // No representation found – the Python version would return `None`
        // and then print an empty line.
        println!();
    }
}
